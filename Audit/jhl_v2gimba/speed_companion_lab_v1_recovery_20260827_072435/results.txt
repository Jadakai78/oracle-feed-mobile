 106:         has_strong_body = _body_fraction(candle) >= IMPULSE_MIN_BODY_FRACTION
 107:         has_large_range = candle_range >= atr * IMPULSE_MIN_ATR
 108: 
 109:         if is_bullish and has_strong_body and has_large_range:
 110:             return {
 111:                 "index": index,
 112:                 "size_atr": candle_range / atr,
 113:             }
 114: 
 115:     return None

 118: def _long_controlled_pullback(
 119:     candles: List[Dict[str, float]],
 120:     impulse_index: int,
 121: ) -> Optional[Dict[str, Any]]:
 122:     """
 123:     Return pullback measurements only when all completed post-impulse bars
 124:     form a bounded, lower-participation, non-bullish pullback.
 125:     """
 126:     impulse = candles[impulse_index]
 127:     pullback = candles[impulse_index + 1:]
 128: 
 129:     if not pullback or len(pullback) > MAX_PULLBACK_BARS:
 130:         return None
 131: 
 132:     if any(candle["close"] > candle["open"] for candle in pullback):
 133:         return None
 134: 
 135:     impulse_origin = impulse["open"]
 136:     impulse_high = impulse["high"]
 137:     impulse_distance = impulse_high - impulse_origin
 138:     if impulse_distance <= 0 or impulse["volume"] <= 0:
 139:         return None
 140: 
 141:     lowest_close = min(candle["close"] for candle in pullback)
 142:     if lowest_close <= impulse_origin:
 143:         return None
 144: 
 145:     pullback_depth_fraction = (impulse_high - lowest_close) / impulse_distance
 146:     if pullback_depth_fraction > MAX_PULLBACK_FRACTION:
 147:         return None
 148: 
 149:     average_pullback_volume = (
 150:         sum(candle["volume"] for candle in pullback) / len(pullback)
 151:     )
 152:     pullback_volume_vs_impulse = (
 153:         average_pullback_volume / impulse["volume"]
 154:     )
 155:     if pullback_volume_vs_impulse > PULLBACK_VOLUME_MAX_FRACTION:
 156:         return None
 157: 
 158:     return {
 159:         "pullback_depth_fraction": round(pullback_depth_fraction, 6),
 160:         "pullback_bars": len(pullback),
 161:         "pullback_volume_vs_impulse": round(
 162:             pullback_volume_vs_impulse,
 163:             6,
 164:         ),
 165:     }

 168: def analyze_completed_candles(
 169:     candles: List[Dict[str, float]],
 170: ) -> Dict[str, Any]:
 171:     """
 172:     Analyze completed OHLCV candles.
 173: 
 174:     V1 behavior:
 175:     - Identify the newest qualifying long impulse.
 176:     - Return CONTROLLED_PULLBACK for a valid bounded pullback.
 177:     - Otherwise return WATCH while the impulse remains in its watch window.
 178:     """
 179:     result = _empty_result()
 180: 
 181:     if len(candles) < ATR_PERIOD + 1:
 182:         return result
 183: 
 184:     impulse = _find_latest_long_impulse(candles)
 185:     if impulse is None:
 186:         return result
 187: 
 188:     impulse_age_bars = len(candles) - 1 - impulse["index"]
 189:     if impulse_age_bars > MAX_WATCH_BARS:
 190:         return result
 191: 
 192:     result["direction"] = "LONG"
 193:     result["impulse_age_bars"] = impulse_age_bars
 194:     result["impulse_size_atr"] = round(impulse["size_atr"], 6)
 195: 
 196:     pullback = _long_controlled_pullback(candles, impulse["index"])
 197:     if pullback is not None:
 198:         result["phase"] = PHASE_CONTROLLED_PULLBACK
 199:         result.update(pullback)
 200:         return result
 201: 
 202:     result["phase"] = PHASE_WATCH
 203:     return result
